cmake_minimum_required(VERSION 3.25)
project(Burnley73 VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BURNLEY73_FETCH_VST3_SDK "Fetch the pinned Steinberg VST3 SDK when no local SDK is supplied" ON)
set(BURNLEY73_VST3_SDK_ROOT "" CACHE PATH "Path to an existing Steinberg VST3 SDK checkout")

set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "Disable VSTGUI" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "Disable VST3 SDK examples" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "Disable VST3 hosting examples" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "Keep plug-in package in the build tree" FORCE)

if(BURNLEY73_VST3_SDK_ROOT STREQUAL "")
    if(NOT BURNLEY73_FETCH_VST3_SDK)
        message(FATAL_ERROR "Set BURNLEY73_VST3_SDK_ROOT or enable BURNLEY73_FETCH_VST3_SDK")
    endif()
    include(FetchContent)
    FetchContent_Declare(vst3sdk
        GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
        GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96
        GIT_SHALLOW FALSE
        GIT_SUBMODULES_RECURSE TRUE
    )
    FetchContent_MakeAvailable(vst3sdk)
    set(BURNLEY73_VST3_SDK_ROOT "${vst3sdk_SOURCE_DIR}")
else()
    if(NOT EXISTS "${BURNLEY73_VST3_SDK_ROOT}/CMakeLists.txt")
        message(FATAL_ERROR "BURNLEY73_VST3_SDK_ROOT is not a VST3 SDK checkout: ${BURNLEY73_VST3_SDK_ROOT}")
    endif()
    add_subdirectory("${BURNLEY73_VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/vst3sdk")
endif()

add_library(burnley73_dsp STATIC src/dsp.cpp src/eq.cpp src/model.cpp src/engine.cpp)
target_include_directories(burnley73_dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(burnley73_dsp PUBLIC cxx_std_17)
set_target_properties(burnley73_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(burnley73_dsp PRIVATE /fp:strict)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
    target_compile_options(burnley73_dsp PRIVATE -ffp-contract=off)
endif()

smtg_add_vst3plugin(Burnley73 src/plugin.cpp src/factory.cpp)
if(WIN32)
    target_sources(Burnley73 PRIVATE "${BURNLEY73_VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp")
elseif(APPLE)
    target_sources(Burnley73 PRIVATE "${BURNLEY73_VST3_SDK_ROOT}/public.sdk/source/main/macmain.cpp")
else()
    target_sources(Burnley73 PRIVATE "${BURNLEY73_VST3_SDK_ROOT}/public.sdk/source/main/linuxmain.cpp")
endif()
target_include_directories(Burnley73 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(Burnley73 PRIVATE cxx_std_17)
target_link_libraries(Burnley73 PRIVATE sdk burnley73_dsp)
if(MINGW)
    target_link_options(Burnley73 PRIVATE -static -static-libgcc -static-libstdc++)
endif()
smtg_target_configure_version_file(Burnley73)
